static void burbuja(int[] a) {
    int n = a.length;
    for (int i = 0; i < n - 1; i++) {
        boolean huboCambio = false;
        for (int j = 0; j < n - 1 - i; j++) {
            if (a[j] > a[j + 1]) {
                int tmp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = tmp;
                huboCambio = true;
            }
        }
        if (!huboCambio) break;
    }
}
